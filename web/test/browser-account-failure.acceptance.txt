async(page)=>{
  let injected=0;
  const handler=async route=>{injected++;await route.fulfill({status:500,contentType:'application/json',body:JSON.stringify({code:50000,message:'验收模拟失败'})});};
  await page.route('**/api/v1/admin/suppliers/8/accounts',handler);
  try{
    await page.getByRole('row').filter({hasText:'FULL1843'}).getByRole('button',{name:'账号管理',exact:true}).click();
    await page.getByText('验收模拟失败',{exact:true}).last().waitFor();
    await page.waitForTimeout(300);
    const spinning=await page.locator('.arco-drawer .arco-spin-loading').count();
    return {injected,spinning,hasRetry:await page.getByRole('button',{name:'重试',exact:true}).count()};
  } finally {await page.unroute('**/api/v1/admin/suppliers/8/accounts',handler);}
}
