async(page)=>{
  await page.goto('http://127.0.0.1:5173/projects');
  await page.getByRole('row').filter({hasText:'LIVE0905165733完整验收486ceb'}).getByRole('button',{name:'进入',exact:true}).click();
  await page.getByRole('button',{name:'撤回',exact:true}).first().waitFor();
  let release,capturedResolve,simulated=0;
  const hold=new Promise(resolve=>release=resolve),captured=new Promise(resolve=>capturedResolve=resolve);
  const handler=async route=>{const response=await route.fetch();capturedResolve();await hold;await route.fulfill({response});};
  const withdraw=async route=>{simulated++;await route.fulfill({status:200,contentType:'application/json',body:'{}'});};
  await page.route('**/api/v1/projects/36',handler);
  await page.route('**/api/v1/projects/36/withdraw',withdraw);
  try{
    await page.getByRole('button',{name:'撤回',exact:true}).first().click();
    await page.getByRole('button',{name:'确定',exact:true}).click();
    await Promise.race([captured,page.waitForTimeout(10000).then(()=>{throw Error('refresh not captured')})]);
    await page.goBack();
    await page.getByRole('row').filter({hasText:'FULL_UI1900'}).getByRole('button',{name:'进入',exact:true}).click();
    await page.getByText('FULL1900项目已编辑',{exact:true}).waitFor();
    release();await page.waitForLoadState('networkidle');
    await page.getByText('FULL1900项目已编辑',{exact:true}).waitFor();
    if(simulated!==1)throw Error('cancel simulation count '+simulated);
    return {latePostOperationRefreshIgnored:true,targetProject:37,simulatedWithdrawResponses:simulated,realProjectStateUnchanged:true};
  }finally{release();await page.unroute('**/api/v1/projects/36',handler);await page.unroute('**/api/v1/projects/36/withdraw',withdraw);}
}
